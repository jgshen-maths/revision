## Arithmetic verification (`/tmp/t005_check.py`)
```python
import math

# T1: smallest n so 240*n is a perfect square, and sqrt(240*n)
assert 2**4 * 3 * 5 == 240
assert 2**3 * 3 * 5 == 120
assert 16 * 15 == 240
n_t1 = 3 * 5
assert n_t1 == 15
assert 3 + 5 == 8
assert 240 * n_t1 == 3600
assert 2**4 * 3**2 * 5**2 == 3600
assert math.isqrt(3600) == 60
assert 2**2 * 3 * 5 == 60
assert 2**1 * 3 * 5 == 30
# verify n=15 is indeed the smallest positive integer
for k in range(1, n_t1):
    assert math.isqrt(240 * k) ** 2 != 240 * k
print("t1 OK: 240 = 2^4*3*5, smallest n = 15, 240*15 = 3600 = 60^2")

# T2: solve 16^x = 8^4
assert 16 == 2**4 and 8 == 2**3
assert 3 * 4 == 12 and 3 + 4 == 7 and 3**4 == 81
x_t2 = 12 // 4
assert x_t2 == 3 and 12 - 4 == 8 and 12 * 4 == 48
assert 16**x_t2 == 8**4 == 4096
assert 8**3 == 512 and 16**2 == 256
print("t2 OK: 16^3 = 8^4 = 2^12 = 4096, x = 3")

# T3: 5 + 36 / (9 - 3) * 2^2 - 7
s1 = 9 - 3
assert s1 == 6
s2 = 2**2
assert s2 == 4
s3 = 36 // s1
assert s3 == 6 and s1 * s2 == 24
s4 = s3 * s2
assert s4 == 24
ans_t3 = 5 + s4 - 7
assert ans_t3 == 22
assert 5 + 36 // (9 - 3) * (2**2) - 7 == 22
assert 5 + 24 + 7 == 36
print("t3 OK: 5 + 36/(9-3)*2^2 - 7 = 5 + 24 - 7 = 22")

# T4: last digit of 18^75 and 18^76
cyc = [pow(8, k, 10) for k in range(1, 6)]
assert cyc == [8, 4, 2, 6, 8]
assert 75 == 4 * 18 + 3
assert pow(18, 75, 10) == pow(8, 75, 10) == cyc[2] == 2
assert 76 == 4 * 19 + 0
assert pow(18, 76, 10) == pow(8, 76, 10) == cyc[3] == 6
print("t4 OK: cycle(8) = [8,4,2,6], 75 = 4*18+3 -> 18^75 ends in 2; 76 = 4*19+0 -> 18^76 ends in 6")
```
